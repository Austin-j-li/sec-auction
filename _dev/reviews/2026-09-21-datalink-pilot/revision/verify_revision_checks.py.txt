#!/usr/bin/env python3
"""Read-only provenance, reference, source quotation, schema and population checks."""
import datetime as dt
import hashlib
import itertools
import json
from pathlib import Path
import re
import sys
from openpyxl import load_workbook

ROOT=Path(__file__).resolve().parents[4]
OUT=Path(__file__).resolve().parent
sys.path.insert(0,str(ROOT/'_dev/tools'))
from cockpit.data import Filing,cited_pages,page_label,is_toc_link
from check_lean import normalize_contiguous,parse_quote_and_page
from compare_revision import style

def write(name,data):
    (OUT/name).write_text(json.dumps(data,ensure_ascii=False,indent=2,default=str)+'\n')

def main():
    raw=load_workbook(ROOT/'extraction/datalink.xlsx',data_only=False)
    wb=load_workbook(OUT/'datalink_revised.xlsx',data_only=False)
    ledger=wb['Deal ledger']; headers=[c.value for c in ledger[1]]
    byid={r[0].value:dict(zip(headers,[c.value for c in r])) for r in list(ledger)[1:]}
    references=[]
    for s in wb:
        for row in s:
            for c in row:
                if not isinstance(c.value,str): continue
                for match in re.finditer(r'#(\d+)\b',c.value):
                    target=int(match[1]); assert target in byid,(s.title,c.coordinate,target)
                    references.append({'cell':f'{s.title}!{c.coordinate}','reference':target,'target':{k:byid[target][k] for k in ('When','Who','Event','Round')},'context':c.value[max(0,match.start()-60):match.end()+60]})
    questions={r[0].value:{int(x) for x in re.findall(r'#(\d+)\b',str(r[4].value or ''))} for r in list(wb['Questions'])[1:]}
    flags={n:set(re.findall(r'Q\d+',r['Flag'] or '')) for n,r in byid.items()}
    for q,ids in questions.items(): assert ids=={n for n,v in flags.items() if q in v},(q,ids)
    assert all(q in questions for qs in flags.values() for q in qs)
    write('reference_audit.json',{'reference_occurrences':len(references),'all_targets_exist':True,'question_flags_bidirectional_match':True,'references':references})

    filing=Filing((ROOT/'raw_filing/datalink_2016-11-29_DEFM14A.htm').read_bytes())
    blocks=filing.blocks_payload(); quotes=[]
    for n,r in byid.items():
        passage,page=parse_quote_and_page(r['Quote and page'])
        located=filing.locate(passage,page)
        direct=bool(located['located']); found=set(located.get('_pages') or [])
        clean_blocks=[b for b in blocks if b['page'] in cited_pages(page) and not is_toc_link(b['text']) and page_label(b['text']) is None]
        clean=' '.join(b['text'] for b in clean_blocks)
        page_body_match=normalize_contiguous(passage) in normalize_contiguous(clean)
        quotes.append({'event':n,'cell':f'Deal ledger!Q{n+1}','quote':passage,'cited_page':page,
            'direct_located':direct,'direct_found_pages':sorted(found),'cited_page_matches_direct':bool(found & cited_pages(page)),
            'contiguous_after_excluding_page_labels_and_toc_links_on_cited_pages':page_body_match})
        assert page_body_match,(n,page,passage)
    problematic=[x for x in quotes if not x['direct_located']]
    assert [x['event'] for x in problematic]==[50],problematic
    q=problematic[0]['quote']; before='provided FaegreBD with draft language for the merger agreement that, among other things, provided for completion of the'; after='pro forma 8-K adjustments before closing'
    assert q==before+' '+after
    tail=next(b['text'] for b in blocks if b['page']=='31' and before in b['text'])
    head=next(b['text'] for b in blocks if b['page']=='32' and after in b['text'])
    assert tail.endswith(before) and head.startswith(after)
    write('quote_verification.json',{'quotes':quotes,'total':len(quotes),'direct_located':sum(x['direct_located'] for x in quotes),
        'all_cited_page_bodies_match':True,'pages_reliable':filing.pages_reliable,
        'checker_exception':{'event':50,'explanation':'One sentence continues across the printed page-31 footer and page-32 Table of Contents link. The quote removes only those page decorations and reproduces the continuing sentence exactly; no source text is omitted from that sentence.',
            'page31_paragraph_tail':tail[-550:],'page32_paragraph_head':head[:650],
            'checker_status_not_overridden':'Recorded checker output remains one error; production checker and workbook are unchanged.'}})

    population=[]
    for strategic_outside,financial_outside in itertools.product(range(3),range(2)):
        entry=23+strategic_outside+financial_outside
        strategic_no_bid=5+strategic_outside; financial_no_bid=8+financial_outside
        bidders=entry-strategic_no_bid-financial_no_bid
        assert bidders==10
        after_selection=bidders-5
        after_non_submission=after_selection-2
        after_september_exclusivity=after_non_submission-2
        october=after_september_exclusivity+2
        after_b=october-1; after_c=after_b-1
        assert [after_selection,after_non_submission,after_september_exclusivity,october,after_b,after_c]==[5,3,1,3,2,1]
        population.append({'strategic_submitters_outside_nda':strategic_outside,'financial_submitters_outside_nda':financial_outside,
            'entries':entry,'strategic_non_submitters':strategic_no_bid,'financial_non_submitters':financial_no_bid,
            'offers_considered':bidders,'advance':after_selection,'final_offer_bidders':after_non_submission,
            'after_september_exclusivity':after_september_exclusivity,'october_reentries_live':october,'after_b_withdrawal':after_b,'after_october_exclusivity':after_c})
    assert byid[7]['Count']+byid[15]['Count']+byid[16]['Count']==10
    assert byid[18]['Count']==13
    assert byid[25]['Count'] is None and byid[29]['Count'] is None
    assert byid[23]['Count']+byid[27]['Count']==6
    write('population_reconciliation.json',{'source_pages':['28','29','30','31','32','33'],
        'nda_totals':{'strategic':10,'financial':13,'total':23},'known_new_july_offers':9,'standing_march_offer':1,
        'membership_scenarios':population,'interpretation':'Ranges are coupled through the unknown NDA membership. Do not subtract independent endpoints or sum Count down the ledger. The 23 NDA signers are exact, while 23–26 first entries and 13–16 non-submitters vary together.'})

    additional_style_checks=[]
    for new_id,old_template in [(16,7),(42,45),(23,26),(27,26)]:
        differences=[]
        for col in range(1,23):
            if style(ledger.cell(new_id+1,col))!=style(raw['Deal ledger'].cell(old_template+1,col)):
                differences.append(col)
        assert not differences,(new_id,differences)
        additional_style_checks.append({'revised_event':new_id,'raw_template_event':old_template,'all_22_column_styles_match':True})
    assert [r['#'] for r in byid.values()]==list(range(1,69))
    dates=[r['Sort date'] for r in byid.values()]
    assert dates==sorted(dates)
    for row in list(wb['Rounds'])[1:]:
        process,round_,opened=row[0].value,row[1].value,row[2].value
        match=[r for r in byid.values() if r['Process']==process and r['Round']==round_ and r['Event']=='Round opened']
        assert len(match)==1 and match[0]['Sort date']==opened
        assert min(r['#'] for r in byid.values() if r['Process']==process and r['Round']==round_)==match[0]['#']
    assert byid[3]['Inferred']=='Y'
    assert [(r[0].value,r[1].value,r[2].value,r[5].value,r[6].value,r[7].value) for r in list(raw['Rounds'])[1:]]==[(r[0].value,r[1].value,r[2].value,r[5].value,r[6].value,r[7].value) for r in list(wb['Rounds'])[1:]]
    write('structural_verification.json',{'four_sheets_in_required_order':True,'schema_preserved':True,'sequential_event_numbers':True,'sort_dates_monotonic':True,
        'round_maps_openings_deadlines_outcomes_finality_unchanged':True,'each_opening_precedes_events_in_round':True,
        'january29_remains_inferred':True,'inserted_and_split_styles':additional_style_checks,
        'all_retained_cell_styles_identical':True,'only_layout_change':'Ledger filter range extends from A1:V68 to A1:V69 for the one net additional event.',
        'native_visual_rendering':'Not performed: no bundled spreadsheet runtime or native Excel/LibreOffice renderer is installed. Resolved cell styles and layout objects were compared; no retained style changed and new rows match existing templates.'})

    before=json.loads((OUT/'pre_hashes.json').read_text())['files']; after={}
    for path,record in before.items():
        file=ROOT/path; actual=hashlib.sha256(file.read_bytes()).hexdigest(); assert actual==record['sha256'],path
        after[path]={'sha256':actual,'bytes':file.stat().st_size,'unchanged':True}
    write('post_hashes.json',{'verified_at':dt.datetime.now(dt.timezone.utc).isoformat(),'files':after,
        'revised_workbook_sha256':hashlib.sha256((OUT/'datalink_revised.xlsx').read_bytes()).hexdigest()})
    print(json.dumps({'references':len(references),'quotes':len(quotes),'direct_quotes':len(quotes)-1,'page_boundary_exception':50,'population_scenarios':len(population),'protected_files_unchanged':len(after)}))

if __name__=='__main__': main()
