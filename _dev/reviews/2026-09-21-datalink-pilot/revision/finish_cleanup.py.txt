#!/usr/bin/env python3
"""Remove only this completed disposable run after its evidence is preserved."""
import datetime as dt
import hashlib
import json
from pathlib import Path
import shutil

ROOT=Path(__file__).resolve().parents[4]
OUT=Path(__file__).resolve().parent
RUN=ROOT/'_dev/runs/datalink-revision-20260921-controlled'

def sha(path): return hashlib.sha256(path.read_bytes()).hexdigest()

def main():
    assert RUN.parent==ROOT/'_dev/runs' and RUN.name=='datalink-revision-20260921-controlled'
    assert json.loads((RUN/'status.json').read_text())['state']=='completed'
    receipt=json.loads((OUT/'run_receipt.json').read_text())
    assert sha(OUT/'datalink_revised.xlsx')==receipt['revised_workbook_sha256']==sha(RUN/'extraction/datalink.xlsx')
    assert sha(OUT/'revision_notes.md')==receipt['revision_notes_sha256']==sha(RUN/'extraction/revision_notes.md')
    for name in ('metadata.json','status.json','command.json','launch.json','validation.json','prompt.txt','stderr.log','launcher.log'):
        assert sha(OUT/('provider_'+name))==sha(RUN/name),name
    assert sha(RUN/'events.jsonl')==receipt['provider_events_sha256']
    for path,record in json.loads((OUT/'pre_hashes.json').read_text())['files'].items(): assert sha(ROOT/path)==record['sha256'],path
    files=[{'path':str(p.relative_to(RUN)),'bytes':p.stat().st_size,'sha256':sha(p)} for p in sorted(RUN.rglob('*')) if p.is_file()]
    removed_helpers=[]
    for name in ('raw_workbook_cells.json','source_pages.txt'):
        path=OUT/name
        removed_helpers.append({'path':name,'bytes':path.stat().st_size,'sha256':sha(path),'reason':'Redundant temporary inspection snapshot; complete change evidence and source_pages.json retained.'})
        path.unlink()
    scripts=[]
    for path in sorted(OUT.glob('*.py')):
        dst=path.with_name(path.name+'.txt')
        assert not dst.exists()
        digest=sha(path); path.rename(dst)
        scripts.append({'original':path.name,'preserved_as':dst.name,'sha256':digest})
    shutil.rmtree(RUN)
    assert not RUN.exists()
    cleanup={'completed_at':dt.datetime.now(dt.timezone.utc).isoformat(),'deleted_run_directory':str(RUN),
        'deleted_run_files':files,'run_directory_absent':True,'provider_state_removed':receipt['temporary_state_removed'],
        'redundant_snapshots_removed':removed_helpers,'administration_and_verification_scripts_preserved_as_evidence':scripts,
        'boundary':'Only the exact assigned Datalink revision run and named redundant snapshots in this revision folder were removed. Other run folders, canonical inputs, original reviews and shared documents were untouched.',
        'no_second_model_pass':True,'no_commit_or_push':True}
    (OUT/'run_cleanup.json').write_text(json.dumps(cleanup,indent=2,ensure_ascii=False)+'\n')
    artifacts={str(p.relative_to(OUT)):{'bytes':p.stat().st_size,'sha256':sha(p)} for p in sorted(OUT.rglob('*')) if p.is_file() and p.name!='artifact_manifest.json'}
    (OUT/'artifact_manifest.json').write_text(json.dumps({'created_at':dt.datetime.now(dt.timezone.utc).isoformat(),'files':artifacts},indent=2,ensure_ascii=False)+'\n')
    print(json.dumps({'run_absent':not RUN.exists(),'artifact_count':len(artifacts),'revised_sha256':sha(OUT/'datalink_revised.xlsx')}))

if __name__=='__main__': main()
